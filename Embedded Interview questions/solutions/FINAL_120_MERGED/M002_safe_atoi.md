```c
#include <stdint.h>
#include <stddef.h>
#include <limits.h>
#include <stdint.h>
#include <stddef.h>
int my_atoi(const char *s,int *out) {
    int sign=1,v=0;
    if(!s||!*s)return -1;
    if(*s=='-') {
        sign=-1;
        ++s;
    }
    while(*s>='0'&&*s<='9') {
        v=v*10+(*s-'0');
        ++s;
    }
    *out=sign*v;
    return 0;
}
typedef enum {
    ATOI_OK = 0, ATOI_NULL = -1, ATOI_EMPTY = -2, ATOI_OVERFLOW = -3, ATOI_INVALID = -4
}
atoi_err_t;
static int skip_ws(const char **s) {
    while (**s == ' ' || **s == '\t' || **s == '\n') {
        (*s)++;
    }
    return 0;
}
atoi_err_t safe_atoi(const char *s, int *out) {
    if (!s || !out) return ATOI_NULL;
    skip_ws(&s);
    if (!*s) return ATOI_EMPTY;
    int sign = 1;
    if (*s == '-') {
        sign = -1;
        s++;
    }
    else if (*s == '+') {
        s++;
    }
    if (*s < '0' || *s > '9') return ATOI_INVALID;
    long v = 0;
    while (*s >= '0' && *s <= '9') {
        v = v * 10 + (*s - '0');
        if ((sign == 1 && v > INT32_MAX) || (sign == -1 && v > (long)INT32_MAX + 1)) {
            return ATOI_OVERFLOW;
        }
        s++;
    }
    *out = (int)(sign * v);
    return ATOI_OK;
}
```
