## Step 3 — Complete solution (compilable C)

```c
#include <stdint.h>
#include <stddef.h>
typedef struct {
    uint32_t overruns, frames;
}
uart_stats_t;
void stat_frame(uart_stats_t *s) {
    s->frames++;
}
#include <stdint.h>
#include <stddef.h>
typedef struct {
    volatile uint32_t SR, DR;
}
spi_t;
int spi_transfer(spi_t *s,const uint8_t *tx,uint8_t *rx,size_t n) {
    for(size_t i=0;
    i<n;
    ++i) {
        while(!(s->SR&(1u<<1)));
        uint8_t out=tx?tx[i]:0xFF;
        s->DR=out;
        while(!(s->SR&(1u<<0)));
        uint8_t in=(uint8_t)s->DR;
        if(rx)rx[i]=in;
    }
    return 0;
}
```
