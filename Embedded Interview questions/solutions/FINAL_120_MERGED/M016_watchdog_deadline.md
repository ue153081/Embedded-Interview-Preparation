#include <stdint.h>
#include <stddef.h>
typedef struct {
    uint32_t last_kick, window_ms;
}
wdt_t;
int wdt_kick(wdt_t *w,uint32_t now) {
    if(now-w->last_kick>w->window_ms)return -1;
    w->last_kick=now;
    return 0;
}

#include <stdint.h>
#include <stddef.h>
typedef struct {
    uint32_t wcet_us;
    uint32_t period_us;
    uint32_t miss_count;
}
task_profile_t;
int sched_feasibility(task_profile_t *t, size_t n, uint32_t frame_us) {
    uint64_t sum = 0;
    for (size_t i = 0;
    i < n;
    ++i) {
        sum += t[i].wcet_us;
    }
    return sum <= frame_us ? 0 : -1;
}
void on_deadline_miss(task_profile_t *t) {
    t->miss_count++;
}
